1. ஹார்ட் டிஸ்க் அன்மவுண்ட் ஆனால் சரிசெய்ய (Fix Unmounted Drive) 
# 1. டிஸ்க் UUID சரிபார்க்கவும்
sudo blkid

# 2. அனைத்து டிரைவ்களையும் மீண்டும் மவுண்ட் செய்யவும்
sudo mount -a

# 3. குறிப்பிட்ட டிரைவை மட்டும் மவுண்ட் செய்ய
sudo mount /dev/sdb1 /mnt/storage

# 4. பழுதடைந்த கோப்புகளை சரிசெய்ய (Filesystem check)
sudo fsck -f /dev/sdb1
2. சர்வர் மெமரி / ஸ்டோரேஜ் 100% நிறைந்தால் காலி செய்ய (100% Disk Full) 
# பழைய கணினி லாக் பைல்களை அழிக்க (Free up GBs immediately)
sudo journalctl --vacuum-time=2d

# தேவையில்லாத டாக்கர் கேச் மற்றும் கண்டெய்னர்களை அழிக்க
sudo docker system prune -af --volumes

# சிஸ்டம் பேக்கேஜ் கேச்சை அழிக்க
sudo apt clean && sudo rm -rf /tmp/*
3. Nginx 502 / இணையதளங்கள் நிற்கும்போது (Nginx 502 Bad Gateway) 
# Nginx கான்ஃபிக் கோப்பில் பிழை உள்ளதா என பார்க்க
sudo nginx -t

# Nginx-ஐ ரீஸ்டார்ட் செய்ய
sudo systemctl restart nginx

# தவறை கண்டறிய லாக் பார்க்க
sudo journalctl -u nginx -n 30 --no-pager
4. Cloudflare Tunnel & Tailscale ரீசெட் செய்ய (Tunnel & VPN Reset) 
# கிளவுட்ஃப்ளேர் டன்னலை ரீஸ்டார்ட் செய்ய

# டெய்ல்ஸ்கேல் நெட்வொர்க்கை ரீசெட் செய்ய
